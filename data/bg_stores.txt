# Z80 instructions that store into the background RAM window $C000-$CFFF
# (bank selected by out ($0D)) other than block moves. They address the
# current bank in bgram directly; tools/harness/amiga_sim reports others
# as BGSTALE. Format: bank (ff = fixed ROM), address.
ff 090d  ld (hl),$00
ff 091f  ld (hl),$00
ff 0931  ld (hl),$00
ff 0943  ld (hl),$00
ff 2af7  ld (de),a
ff 2afb  ld (de),a
ff 5de4  ld (de),a	level change: the next level's background patches ($8D90, bank 4)
ff 5de8  ld (de),a
02 b951  ld (de),a	a level 3 object changing the scenery
02 b955  ld (de),a
02 b95f  ld (de),a
02 b963  ld (de),a
